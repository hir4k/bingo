// The application owns its assets and registration; Bingo dispatches runtime commands.
package main

import (
 "embed"
 "log"
 "os"

 "__MODULE__/config"
 "github.com/hir4k/bingo"
)

// all: includes .keep files so an empty application can still embed these folders.
//go:embed all:views all:database/migrations
var assets embed.FS

func main() {
 var app *bingo.App=bingo.New()
 app.Views(assets)
 app.Database(config.Database)
 app.Routes(config.RegisterRoutes)
 app.Commands(config.RegisterCommands)
 app.Jobs(config.RegisterJobs)
 var address string=os.Getenv("BINGO_ADDR")
 if address=="" { address=":8080" }
 if err:=app.Run(address);err!=nil {log.Fatal(err)}
}
